METHOD wordCounter(text):
    words = SPLIT text into list
    map = EMPTY MAP<String, Integer>

    FOR each word in words:
        word = LOWERCASE(word)
        REMOVE punctuation
        IF word in map THEN map[word] += 1
        ELSE map[word] = 1

    RETURN map

MAIN:
    text = "Sports bring people together. Sports inspire teamwork. Sports create leaders."
    result = wordCounter(text)
    PRINT result
