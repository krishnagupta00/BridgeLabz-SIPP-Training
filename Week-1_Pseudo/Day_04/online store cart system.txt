class Item {
    attribute name
    attribute price
    attribute quantity

    constructor(name, price, quantity)
}

class Cart {
    attribute items : List<Item>

    constructor() {
        items = empty list
    }

    method addItem(item) {
        items.add(item)
    }

    generic method calculateTotal(list) {
        total = 0
        for each element in list
            total = total + (element.price * element.quantity)
        return total
    }
}

main {
    cart = new Cart()

    cart.addItem(new Item("Laptop", 50000, 1))
    cart.addItem(new Item("Mouse", 800, 2))
    cart.addItem(new Item("Keyboard", 1500, 1))

    total = cart.calculateTotal(cart.items)
    print "Total Price: " + total
}
